import cv2
from deepface import DeepFace
import numpy as np

# Load OpenCV's built-in pre-trained Haar Cascade Face Detector
face_cascade = cv2.CascadeClassifier(
    cv2.data.haarcascades + "haarcascade_frontalface_default.xml"
)

anchor_embedding = None
COSINE_THRESHOLD = 0.40


def extract_face_embedding(crop):
    """Generates facial embedding via FaceNet."""
    try:
        results = DeepFace.represent(
            img_path=crop,
            model_name="Facenet",
            enforce_detection=False,
            detector_backend="skip",
        )
        if results and len(results) > 0:
            return np.array(results[0]["embedding"])
    except Exception:
        pass
    return None


def cosine_distance(vec_a, vec_b):
    dot = np.dot(vec_a, vec_b)
    norm = np.linalg.norm(vec_a) * np.linalg.norm(vec_b)
    return 1.0 - (dot / (norm + 1e-7))


def apply_gaussian_blur(frame, x, y, w, h):
    """Blurs the target bounding box directly on the frame buffer."""
    img_h, img_w, _ = frame.shape
    x1, y1 = max(0, x), max(0, y)
    x2, y2 = min(img_w, x + w), min(img_h, y + h)

    region = frame[y1:y2, x1:x2]
    if region.size == 0:
        return

    # Proportional blur kernel
    kw = (w // 3) * 2 + 1
    kh = (h // 3) * 2 + 1
    kw, kh = max(15, kw), max(15, kh)

    blurred = cv2.GaussianBlur(region, (kw, kh), 30)
    frame[y1:y2, x1:x2] = blurred


cap = cv2.VideoCapture(0)

while cap.isOpened():
    ret, frame = cap.read()
    if not ret:
        break

    frame = cv2.flip(frame, 1)
    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    # Detect faces in frame
    faces = face_cascade.detectMultiScale(
        gray, scaleFactor=1.2, minNeighbors=5, minSize=(60, 60)
    )

    for x, y, w, h in faces:
        face_crop = frame[y : y + h, x : x + w]

        if face_crop.size == 0:
            continue

        if anchor_embedding is None:
            # Unenrolled state: prompt the user
            cv2.rectangle(frame, (x, y), (x + w, y + h), (0, 255, 255), 2)
            cv2.putText(
                frame,
                "Press 'C' to Enroll",
                (x, max(25, y - 10)),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.6,
                (0, 255, 255),
                2,
            )
        else:
            current_embedding = extract_face_embedding(face_crop)
            if current_embedding is not None:
                dist = cosine_distance(anchor_embedding, current_embedding)
                if dist <= COSINE_THRESHOLD:
                    # Verified Primary User
                    cv2.rectangle(frame, (x, y), (x + w, y + h), (0, 255, 0), 2)
                    cv2.putText(
                        frame,
                        "User Verified",
                        (x, max(25, y - 10)),
                        cv2.FONT_HERSHEY_SIMPLEX,
                        0.6,
                        (0, 255, 0),
                        2,
                    )
                else:
                    # Bystander: Apply blur
                    apply_gaussian_blur(frame, x, y, w, h)
                    cv2.putText(
                        frame,
                        "Bystander Blurred",
                        (x, max(25, y - 10)),
                        cv2.FONT_HERSHEY_SIMPLEX,
                        0.6,
                        (0, 0, 255),
                        2,
                    )
            else:
                apply_gaussian_blur(frame, x, y, w, h)

    cv2.imshow("PrivaStream Pipeline", frame)

    key = cv2.waitKey(1) & 0xFF
    if key == ord("c"):
        if len(faces) > 0:
            # Sort by area to enroll the primary/closest face
            largest_face = max(faces, key=lambda b: b[2] * b[3])
            fx, fy, fw, fh = largest_face
            face_crop = frame[fy : fy + fh, fx : fx + fw]
            anchor_embedding = extract_face_embedding(face_crop)
            if anchor_embedding is not None:
                print("[SUCCESS] Anchor Identity Enrolled.")
    elif key == ord("r"):
        anchor_embedding = None
        print("[RESET] Anchor Identity Cleared.")
    elif key == ord("q"):
        break

cap.release()
cv2.destroyAllWindows()
